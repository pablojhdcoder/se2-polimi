% Define teal color scheme
\definecolor{tealprimary}{rgb}{0.0, 0.5, 0.5}
\definecolor{tealdark}{rgb}{0.0, 0.35, 0.35}

% Custom theorem environments
\declaretheoremstyle[
    headfont=\color{tealprimary}\normalfont\bfseries,
    bodyfont=\color{black}\normalfont\itshape,
]{colored}

% Set-up caption colors
\captionsetup[figure]{labelfont={color=tealprimary}} % Set colour of the captions
\captionsetup[table]{labelfont={color=tealprimary}} % Set colour of the captions
\captionsetup[algorithm]{labelfont={color=tealprimary}} % Set colour of the captions

\theoremstyle{colored}
\newtheorem{theorem}{Theorem}[chapter]
\newtheorem{proposition}{Proposition}[chapter]

% Enhances the features of the standard "table" and "tabular" environments.
\newcommand\T{\rule{0pt}{2.6ex}}
\newcommand\B{\rule[-1.2ex]{0pt}{0pt}}

% Pseudo-code algorithm descriptions.
\newcounter{algsubstate}
\renewcommand{\thealgsubstate}{\alph{algsubstate}}
\newenvironment{algsubstates}
{\setcounter{algsubstate}{0}%
\renewcommand{\STATE}{%
    \stepcounter{algsubstate}%
    \Statex {\small\thealgsubstate:}\space}}
{}

% New font size
\newcommand\numfontsize{\@setfontsize\Huge{200}{60}}

% Title format: chapter
\titleformat{\chapter}[hang]{
    \fontsize{50}{20}\selectfont\bfseries\filright}{\textcolor{tealprimary} \thechapter\hsp\quad}{0pt}{\huge\bfseries \textcolor{tealprimary}
}

% Title format: section
\titleformat{\section}
{\color{tealprimary}\normalfont\Large\bfseries}
{\color{tealdark}\thesection}{1em}{}

% Title format: subsection
\titleformat{\subsection}
{\color{tealprimary}\normalfont\large\bfseries}
{\color{tealdark}\thesubsection}{1em}{}

% Title format: subsubsection
\titleformat{\subsubsection}
{\color{tealprimary}\normalfont\normalsize\bfseries}
{\color{tealdark}\thesubsubsection}{1em}{}

% Shortening for setting no horizontal-spacing
\newcommand{\hsp}{\hspace{0pt}}

\makeatletter
% Renewcommand: cleardoublepage including the background pic
\renewcommand*\cleardoublepage{%
    \clearpage\if@twoside\ifodd\c@page\else
    \null
    \AddToShipoutPicture*{\BackgroundPic}
    \thispagestyle{empty}%
    \newpage
    \if@twocolumn\hbox{}\newpage\fi\fi\fi}
\makeatother

%For correctly numbering algorithms
\numberwithin{algorithm}{chapter}
